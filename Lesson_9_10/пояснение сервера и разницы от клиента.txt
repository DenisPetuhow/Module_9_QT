1. Анализ Сервера (TCPServer)
Изучив файлы structs.h и tcpserver.cpp, мы видим следующие жесткие правила протокола:

Выравнивание памяти: Используется #pragma pack(push, 1). Это значит, что в структуре ServiceHeader нет "дырок" (padding). Размер структуры — ровно сумма размеров полей (2+2+1+4 = 9 байт). Клиент обязан использовать такую же директиву.

Магическое число: Все пакеты должны начинаться с ID = 0xFFFE. Если клиент пришлет что-то другое, сервер начнет искать это число в потоке, отбрасывая данные.

Сериализация: Сервер использует QDataStream.

Для строк (QString) Qt автоматически добавляет 4 байта длины перед строкой.

Для структур перегружены операторы << и >>.

Логика SET_DATA: Сервер проверяет поле header.len. Он сравнивает его с freeSpace. Значит, клиент должен очень точно указывать размер отправляемых данных в заголовке до самой отправки.

2. Реализация Клиента (TCPClient)
Ниже представлен полный код клиента. Я привел его к единому стилю (__member, m_arg), добавил защиту от фрагментации TCP и комментарии.

Файл: tcpclient.h
Описание класса-клиента. Он наследуется от QObject, чтобы работать с сигналами и слотами.

C++
#ifndef TCPCLIENT_H
#define TCPCLIENT_H

#include <QObject>
#include <QTcpSocket>
#include <QDataStream>
#include <QDateTime>
#include <QHostAddress>
#include "structs.h" // Обязательно подключаем общий файл с описанием протокола

class TCPclient : public QObject
{
    Q_OBJECT
public:
    explicit TCPclient(QObject *m_parent = nullptr);
    
    // --- Управление соединением ---
    /*!
     * @brief Подключиться к удаленному хосту
     */
    void ConnectToHost(QHostAddress m_host, uint16_t m_port);

    /*!
     * @brief Разорвать соединение
     */
    void DisconnectFromHost(void);
    
    // --- Методы отправки ---
    /*!
     * @brief Отправка сервисного запроса (без полезной нагрузки)
     * Используется для GET_TIME, GET_STAT, CLEAR_SPACE
     */
    void SendRequest(ServiceHeader m_head);

    /*!
     * @brief Отправка строковых данных (SET_DATA)
     * Рассчитывает реальный размер данных для корректного заголовка
     */
    void SendData(ServiceHeader m_head, QString m_str);

private slots:
    /*!
     * @brief Слот, вызываемый Qt, когда в сокете появляются новые данные
     */
    void ReadyRead(void);

private:
    QTcpSocket* __socket;          // Объект сокета для связи
    ServiceHeader __servHeader;    // Буфер для хранения заголовка текущего пакета
    
    /*!
     * @brief Метод обработки полностью принятого пакета
     */
    void ProcessingData(ServiceHeader m_header, QDataStream &m_stream);

signals:
    // Сигналы для уведомления GUI
    void sig_connectStatus(uint16_t m_status);
    void sig_Disconnected(void);
    void sig_Error(uint16_t m_code);
    
    // Сигналы с полезными данными
    void sig_sendTime(QDateTime m_time);
    void sig_sendStat(StatServer m_stat);
    void sig_sendFreeSize(uint32_t m_size);
    void sig_SendReplyForSetData(QString m_reply);
    void sig_Success(uint16_t m_type);
};

#endif // TCPCLIENT_H

C++
#include "tcpclient.h"

/*
 * --- ПЕРЕГРУЗКА ОПЕРАТОРОВ ---
 * Это необходимо, чтобы QDataStream знал, как превращать наши структуры в байты.
 * Порядок полей СТРОГО важен и должен совпадать с structs.h
 */

QDataStream &operator >>(QDataStream &out, ServiceHeader &data){
    out >> data.id;
    out >> data.idData;
    out >> data.status;
    out >> data.len;
    return out;
};

QDataStream &operator <<(QDataStream &in, ServiceHeader &data){
    in << data.id;
    in << data.idData;
    in << data.status;
    in << data.len;
    return in;
};

// Реализуем чтение статистики (на сервере был только <<, клиенту нужен >>)
QDataStream &operator >>(QDataStream &out, StatServer &data){
    out >> data.incBytes;
    out >> data.sendBytes;
    out >> data.revPck;
    out >> data.sendPck;
    out >> data.workTime;
    out >> data.clients;
    return out;
}

// --- РЕАЛИЗАЦИЯ КЛАССА ---

TCPclient::TCPclient(QObject *m_parent) : QObject(m_parent)
{
    __socket = new QTcpSocket(this);

    // Подписываемся на события сокета
    connect(__socket, &QTcpSocket::readyRead, this, &TCPclient::ReadyRead);

    connect(__socket, &QTcpSocket::connected, this, [&]{
        emit sig_connectStatus(STATUS_SUCCES);
    });
    
    // Обработка ошибок соединения
    connect(__socket, &QTcpSocket::errorOccurred, this, [&]{
        emit sig_connectStatus(ERR_CONNECT_TO_HOST);
    });

    connect(__socket, &QTcpSocket::disconnected, this, &TCPclient::sig_Disconnected);
}

void TCPclient::ConnectToHost(QHostAddress m_host, uint16_t m_port)
{
    __socket->connectToHost(m_host, m_port);
}

void TCPclient::DisconnectFromHost()
{
    __socket->disconnectFromHost();
}

void TCPclient::SendRequest(ServiceHeader m_head)
{
    QByteArray buffer;
    QDataStream out(&buffer, QIODevice::WriteOnly);

    m_head.len = 0; // Для запросов типа "Дай время" данных нет
    out << m_head;  // Записываем заголовок в поток

    __socket->write(buffer); // Отправляем байты в сеть
}

void TCPclient::SendData(ServiceHeader m_head, QString m_str)
{
    QByteArray buffer;
    QDataStream out(&buffer, QIODevice::WriteOnly);

    /* * ХИТРОСТЬ С QDataStream:
     * Когда мы пишем QString в поток (out << str), Qt добавляет 4 байта длины перед строкой.
     * Сервер проверяет m_head.len перед чтением. Нам нужно узнать ТОЧНЫЙ размер посылки.
     * Для этого запишем строку во временный буфер.
     */
    QByteArray tempBuffer;
    QDataStream tempOut(&tempBuffer, QIODevice::WriteOnly);
    tempOut << m_str;

    // Теперь мы знаем, сколько байт займет строка вместе с префиксом длины
    m_head.len = tempBuffer.size();

    out << m_head; // 1. Пишем заголовок с правильной длиной
    out << m_str;  // 2. Пишем саму строку (Qt сам добавит префикс длины)

    __socket->write(buffer);
}

/*
 * Самый важный метод. Обрабатывает фрагментацию TCP.
 * Данные могут прийти кусками (например, половина заголовка сейчас, половина через 5 мс).
 */
void TCPclient::ReadyRead()
{
    QDataStream incStream(__socket);

    if(incStream.status() != QDataStream::Ok) return;

    // Цикл нужен, чтобы обработать случаи, когда пришло СРАЗУ НЕСКОЛЬКО пакетов (склейка)
    while(true){

        // ШАГ 1: Читаем ЗАГОЛОВОК
        // Если idData == 0, считаем, что мы ждем новый пакет
        if(__servHeader.idData == 0){

            // Проверяем, набралось ли достаточно байт для заголовка
            if(__socket->bytesAvailable() < sizeof(ServiceHeader)){
                return; // Байт мало, ждем следующего сигнала readyRead
            }

            incStream >> __servHeader;

            // Валидация: проверяем ID пакета
            if(__servHeader.id != ID){
                // Если пришел мусор, пытаемся найти начало следующего пакета
                uint16_t hdr = 0;
                while(__socket->bytesAvailable()){
                     incStream >> hdr;
                     if(hdr == ID) {
                         // Нашли! Читаем остальные поля
                         incStream >> __servHeader.idData; 
                         incStream >> __servHeader.status; 
                         incStream >> __servHeader.len;
                         break;
                     }
                }
            }
        }

        // ШАГ 2: Читаем ДАННЫЕ (Тело сообщения)
        // Проверяем, пришли ли все данные, обещанные в заголовке
        if(__socket->bytesAvailable() < __servHeader.len){
            return; // Данные еще не пришли полностью. Ждем.
        }

        // Если мы здесь — у нас есть Заголовок + Все данные. Обрабатываем!
        ProcessingData(__servHeader, incStream);

        // Сбрасываем состояние, чтобы на следующей итерации читать НОВЫЙ заголовок
        __servHeader.idData = 0;
        __servHeader.status = 0;
        __servHeader.len = 0;
    }
}

void TCPclient::ProcessingData(ServiceHeader m_header, QDataStream &m_stream)
{
    // Сначала проверяем статус ответа от сервера
    if(m_header.status != STATUS_SUCCES){
        emit sig_Error(m_header.status);
        // Если с ошибкой пришли данные (например, текст ошибки), их надо вычитать, чтобы очистить буфер
        if(m_header.len > 0) {
            m_stream.skipRawData(m_header.len);
        }
        return;
    }

    // Разбор полезной нагрузки
    switch (m_header.idData){
        case GET_TIME: {
            QDateTime time;
            m_stream >> time;
            emit sig_sendTime(time);
            break;
        }
        case GET_STAT: {
            StatServer stat;
            m_stream >> stat; // Используем наш перегруженный оператор >>
            emit sig_sendStat(stat);
            break;
        }
        case GET_SIZE: {
            uint32_t freeSpace;
            m_stream >> freeSpace;
            emit sig_sendFreeSize(freeSpace);
            break;
        }
        case SET_DATA: {
            QString reply;
            m_stream >> reply; // Читаем эхо-ответ сервера
            emit sig_SendReplyForSetData(reply);
            break;
        }
        case CLEAR_SPACE: {
            emit sig_Success(CLEAR_SPACE);
            break;
        }
    }
}

#include "mainwindow.h"
#include "./ui_mainwindow.h"

MainWindow::MainWindow(QWidget *m_parent)
    : QMainWindow(m_parent)
    , __ui(new Ui::MainWindow)
{
    __ui->setupUi(this);

    __client = new TCPclient(this);

    // Блокируем кнопки до подключения (защита от дурака)
    __ui->le_data->setEnabled(false);
    __ui->pb_request->setEnabled(false);
    __ui->lb_connectStatus->setText("Отключено");
    __ui->lb_connectStatus->setStyleSheet("color: red");

    // --- Блок Connect (Связываем сигналы) ---

    // 1. Статусы подключения
    connect(__client, &TCPclient::sig_connectStatus, this, &MainWindow::DisplayConnectStatus);
    connect(__client, &TCPclient::sig_Disconnected, this, [&]{
        DisplayConnectStatus(ERR_CONNECT_TO_HOST); // При разрыве используем логику ошибки
    });

    // 2. Ошибки протокола
    connect(__client, &TCPclient::sig_Error, this, &MainWindow::DisplayError);

    // 3. Данные
    connect(__client, &TCPclient::sig_sendTime, this, &MainWindow::DisplayTime);
    connect(__client, &TCPclient::sig_sendStat, this, &MainWindow::DisplayStat);
    connect(__client, &TCPclient::sig_sendFreeSize, this, &MainWindow::DisplayFreeSpace);
    connect(__client, &TCPclient::sig_SendReplyForSetData, this, &MainWindow::SetDataReply);
    connect(__client, &TCPclient::sig_Success, this, &MainWindow::DisplaySuccess);

    // Кнопка очистки лога (простая связь)
    connect(__ui->pb_clear, &QPushButton::clicked, __ui->tb_result, &QTextBrowser::clear);
}

MainWindow::~MainWindow()
{
    delete __ui;
}

// --- Обработчики кнопок (UI) ---

void MainWindow::on_pb_connect_clicked()
{
    if(__ui->pb_connect->text() == "Подключиться"){
        // Собираем IP из 4-х спинбоксов
        QString ip = QString("%1.%2.%3.%4")
                     .arg(__ui->spB_ip4->value())
                     .arg(__ui->spB_ip3->value())
                     .arg(__ui->spB_ip2->value())
                     .arg(__ui->spB_ip1->value());

        uint16_t port = __ui->spB_port->value();
        
        __client->ConnectToHost(QHostAddress(ip), port);
    }
    else{
        __client->DisconnectFromHost();
    }
}

void MainWindow::on_pb_request_clicked()
{
    ServiceHeader header;
    header.id = ID;
    header.status = STATUS_SUCCES;

    // Индексы ComboBox (0-Time, 1-Space, 2-Stat, 3-Send, 4-Clear)
    switch (__ui->cb_request->currentIndex()){
        case 0:
            header.idData = GET_TIME;
            __client->SendRequest(header);
            break;
        case 1:
            header.idData = GET_SIZE;
            __client->SendRequest(header);
            break;
        case 2:
            header.idData = GET_STAT;
            __client->SendRequest(header);
            break;
        case 3:
            header.idData = SET_DATA;
            // Для отправки текста используем специальный метод
            __client->SendData(header, __ui->le_data->text());
            break;
        case 4:
            header.idData = CLEAR_SPACE;
            __client->SendRequest(header);
            break;
        default:
            __ui->tb_result->append("Неизвестная команда");
    }
}

void MainWindow::on_cb_request_currentIndexChanged(int m_index)
{
    // Включаем поле ввода только для "Отправить данные" (индекс 3)
    if(m_index == 3) __ui->le_data->setEnabled(true);
    else __ui->le_data->setEnabled(false);
}

4. Сравнение Клиента и Сервера
Главная разница между ними заключается в их роли в установке соединения и управлении ресурсами.

Характеристика	TCP Сервер (tcpserver.cpp)	TCP Клиент (tcpclient.cpp)
Инициализация	Пассивная. Использует listen(). Он сидит и "слушает" порт, ожидая, пока кто-то постучится.	Активная. Использует connectToHost(). Он знает IP и Порт и сам инициирует "звонок".
Множественность	Один ко Многим. Хранит список подключенных сокетов (QMap<QTcpSocket*, ...>). Он должен помнить состояние для каждого клиента отдельно.	Один к Одному. У клиента всего один сокет (__socket), который связывает его с одним сервером.
Обработка данных	Сложная. Использует очередь (queue) для каждого клиента, чтобы собирать фрагментированные пакеты от множества источников, не смешивая их.	Чуть проще. Читает данные последовательно из одного потока. Но всё равно должен бороться с фрагментацией (цикл while в ReadyRead).
Сетевая адресация	Обычно слушает QHostAddress::Any (принимает подключения с любой сетевой карты).	Должен знать конкретный IP адрес сервера.
Жизненный цикл	Работает постоянно, пока не выключат. При ошибке одного клиента сервер продолжает работать с другими.	Работает сеансами. При разрыве соединения часто требует действий пользователя для переподключения.
Метафора:

Сервер — это оператор колл-центра. Он сидит на месте (Listen) и может говорить с несколькими людьми по очереди или параллельно, удерживая контекст разговора с каждым.

Клиент — это человек с телефоном. Он набирает номер (Connect) и говорит только с оператором.
